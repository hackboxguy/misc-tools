# Local source (host-side clone, private repo): the sources-stage checkout
# revision is part of the apps-stage stamp, so a push to the repo still
# re-triggers the stage automatically.
packages/qt-cluster-demo-hook.sh|file://${REPOBINS}/qt-cluster-demo|local|/home/pi/qt-cluster-demo

# CAN proxy (public repo, in-chroot clone) and its bench vehicle. Both
# install as enabled services with the bench default (emulator --car=ev on
# vcan1, proxy plugin emu-ev on vcan0). The cluster keeps its own default
# above; see car-can-proxy-hook.sh for switching the image to the proxy path.
packages/car-can-proxy-hook.sh|https://github.com/hackboxguy/car-can-proxy.git|main|/home/pi/car-can-proxy|
packages/car-can-emulator-hook.sh|https://github.com/hackboxguy/car-can-emulator.git|main|/home/pi/car-can-emulator|

# Stream Deck: three theme radios (legacy / ev / harman), map-backdrop and
# camera-view toggles -- all of which rewrite qt-cluster-demo.env and restart
# the unit, so a choice survives a reboot -- plus a video key that starts
# cluster-video.service instead, and deliberately does not persist. The screen
# name is the third field. The service is udev-triggered, so this is inert
# until a deck is plugged in.
packages/streamdeck-ctrl-hook.sh|https://github.com/hackboxguy/streamdeck-ctrl.git|qt-cluster-demo|/home/pi/streamdeck-ctrl|socat

# Reference video for the Stream Deck playback key. Private repo, cloned from
# the build host over file:// so no credential enters the chroot; the cmake
# args field selects this board's 29 MB clip instead of micropanel's 89 MB one.
# Installs to /home/pi/media/videos/ (cluster-video.service reads it there).
../../custom-pi-imager/packages/generic-package-hook.sh|file://${REPOBINS}/media-files|local|/home/pi/media|||-DMEDIA_PROFILE=cluster

#eth0: DHCP with static fallback 192.168.10.3/24 (NetworkManager two-profile)
packages/network-fallback-hook.sh

# Keep the journal across reboots. Stock RaspiOS is Storage=volatile, which

packages/persistent-journal-hook.sh
